cmake_minimum_required(VERSION 3.20)
project(DeterministicMediaEnhancer VERSION 0.9.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(dme src/main.cpp)
target_compile_options(dme PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS dme RUNTIME DESTINATION bin)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
enable_testing()
add_test(NAME reproducibility COMMAND bash "${CMAKE_SOURCE_DIR}/tests/reproducibility.sh" "$<TARGET_FILE:dme>")
add_executable(analysis-test tests/analysis.cpp)
target_compile_options(analysis-test PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME analysis-metrics COMMAND analysis-test)
add_test(NAME adaptive-reproducibility COMMAND bash "${CMAKE_SOURCE_DIR}/tests/adaptive.sh" "$<TARGET_FILE:dme>")
add_executable(optimizer-test tests/optimizer.cpp)
target_compile_options(optimizer-test PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME optimizer-guards COMMAND optimizer-test)
add_test(NAME optimizer-integration COMMAND bash "${CMAKE_SOURCE_DIR}/tests/optimizer_integration.sh" "$<TARGET_FILE:dme>")
add_executable(output-guard-test tests/output_guard.cpp)
target_compile_options(output-guard-test PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME full-image-guard COMMAND output-guard-test)
add_test(NAME restoration-integration COMMAND bash "${CMAKE_SOURCE_DIR}/tests/restoration.sh" "$<TARGET_FILE:dme>")
add_executable(video-test tests/video.cpp)
target_compile_options(video-test PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME video-timeline COMMAND video-test)
add_test(NAME video-integration COMMAND bash "${CMAKE_SOURCE_DIR}/tests/video.sh" "$<TARGET_FILE:dme>")
add_executable(multiframe-test tests/multiframe.cpp)
target_compile_options(multiframe-test PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME multiframe-fusion COMMAND multiframe-test)
add_test(NAME multiframe-integration COMMAND bash "${CMAKE_SOURCE_DIR}/tests/multiframe.sh" "$<TARGET_FILE:dme>")
add_test(NAME agent-json-interface COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/agent.py" "${CMAKE_SOURCE_DIR}/scripts/agent.py" "$<TARGET_FILE:dme>")
add_test(NAME batch-json-interface COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/batch.py" "${CMAKE_SOURCE_DIR}/scripts/agent.py" "$<TARGET_FILE:dme>")
add_test(NAME agent-hardening COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/hardening.py" "${CMAKE_SOURCE_DIR}/scripts/agent.py" "$<TARGET_FILE:dme>")
