cmake_minimum_required(VERSION 3.20)
project(DeterministicMediaEnhancer VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(dme src/main.cpp)
target_compile_options(dme PRIVATE -Wall -Wextra -Wpedantic)
find_package(Qt6 COMPONENTS Widgets QUIET)
if(Qt6_FOUND)
  add_executable(dme-gui src/gui.cpp)
  target_link_libraries(dme-gui PRIVATE Qt6::Widgets)
  add_dependencies(dme-gui dme)
endif()
enable_testing()
add_test(NAME reproducibility COMMAND bash "${CMAKE_SOURCE_DIR}/tests/reproducibility.sh" "$<TARGET_FILE:dme>")
